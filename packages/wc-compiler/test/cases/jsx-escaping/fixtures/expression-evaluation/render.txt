render() {
  let reads = 0;
  const readText = () => `Value ${++reads}`;
  const values = {
    get label() {
      return readText();
    }
  };
  this.innerHTML = `<section title="${ExpressionEvaluation.$$escapeHtml(`${readText()}`, true)}" data-reads="${ExpressionEvaluation.$$escapeHtml(`${++reads}`, true)}">
        <p>${ExpressionEvaluation.$$wccChild(readText())}</p>
        <p>${ExpressionEvaluation.$$wccChild(values.label)}</p>
      </section>`;
}
