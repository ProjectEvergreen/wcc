class First extends HTMLElement {
  static $$wccChild(value) {
    if (value == null || typeof value === 'boolean') return '';
    if (Array.isArray(value)) return value.map(First.$$wccChild).join('');
    if (typeof value === 'object' && value[Symbol.for('wc-compiler.jsx-markup')] === true) return value.html;
    return First.$$escapeHtml(`${value}`);
  }
  static $$escapeHtml = (value, attribute) => {
    value = value.replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;');
    return attribute ? value.replaceAll('"', '&quot;') : value;
  };
  connectedCallback() {
    this.render();
  }
  render() {
    this.innerHTML = `<ul>
        ${['first'].map(text => `<li>${First.$$wccChild(text)}</li>`).join('')}
      </ul>`;
  }
}
