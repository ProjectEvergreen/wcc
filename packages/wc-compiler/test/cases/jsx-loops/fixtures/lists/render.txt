render() {
  const items = [{
    name: '<Alex>',
    show: true
  }, {
    name: 'Sam & Jo',
    show: true
  }];
  const saved = items.map(({name}, index) => {
    const title = `${index}: ${name}${this.#suffix}`;
    return $$wcc_Markup(`<li title="${Lists.$$escapeHtml(`${title}`, true)}">
          <span>${Lists.$$wcc_Child(title)}</span>
        </li>`);
  });
  const collected = [];
  items.forEach(({name}) => collected.push($$wcc_Markup(`<em>${Lists.$$wcc_Child(name)}</em>`)));
  const $evBindings = [];
  const $evRegister = handlers => $evBindings.push(handlers) - 1;
  this.innerHTML = `<main>
        <ul id="saved">${Lists.$$wcc_Child(saved)}</ul>
        <ul id="chain">
          ${items.filter(item => item.show).map(({name}) => `<li>${Lists.$$wcc_Child(name)}</li>`).join('')}
        </ul>
        <div id="flat">${items.flatMap(({name}) => Lists.$$wcc_Child([$$wcc_Markup(`<span>${Lists.$$wcc_Child(name)}</span>`), $$wcc_Markup(`<small>Details</small>`)])).join('')}</div>
        <div id="nested">
          ${[1, 2].map(number => `<section>
              ${[3, 4].map(value => `<small>${Lists.$$wcc_Child(number * value)}</small>`).join('')}
            </section>`).join('')}
        </div>
        <div id="empty">
          ${[].map(value => `<p>${Lists.$$wcc_Child(value)}</p>`).join('')}
        </div>
        <div id="foreach">
          ${Lists.$$wcc_Child(items.forEach(({name}) => $$wcc_Markup(`<p>${Lists.$$wcc_Child(name)}</p>`)))}
        </div>
        <div id="collected">${Lists.$$wcc_Child(collected)}</div>
        <div id="helper">${Lists.$$wcc_Child(helper('chosen'))}</div>
        <div id="collision">${Lists.$$wcc_Child($$wccMarkup)}</div>
        <div id="inline">
          ${items.map(({name}) => `<button data-wcc-ev-f3a602132f13-0="${$evRegister([['click', () => this.action()]])}">${Lists.$$wcc_Child(name)}</button>`).join('')}
        </div>
        <table>
          <tbody>
            ${items.map(({name}) => `<tr>
                <td>
                  <button data-wcc-ev-f3a602132f13-0="${$evRegister([['click', event => this.action(event)]])}">${Lists.$$wcc_Child(name)}</button>
                </td>
              </tr>`).join('')}
          </tbody>
        </table>
        <div id="json">
          ${items.map(item => `<span data-comment="${Lists.$$escapeHtml(`${JSON.stringify(item)}`, true)}"></span>`).join('')}
        </div>
      </main>`;
  for (const $evEl of this.querySelectorAll("[data-wcc-ev-f3a602132f13-0]") ?? []) {
    if ($evEl.parentElement?.closest(this.localName) !== this) continue;
    const $evHandlers = $evBindings[Number($evEl.getAttribute("data-wcc-ev-f3a602132f13-0"))];
    for (const [$evName, $evHandler] of $evHandlers ?? []) {
      $evEl.addEventListener($evName, $evHandler);
    }
    $evEl.removeAttribute("data-wcc-ev-f3a602132f13-0");
  }
}
