static $$wccChild(value) {
  if (value == null || typeof value === 'boolean') return '';
  if (Array.isArray(value)) return value.map(MixedChildren.$$wccChild).join('');
  if (typeof value === 'object' && value[Symbol.for('wc-compiler.jsx-markup')] === true) return value.html;
  return `${value}`.replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;');
}
