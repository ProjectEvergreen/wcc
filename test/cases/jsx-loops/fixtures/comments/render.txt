render() {
  const {open, visibility, label} = this;
  const replies = [{
    author: 'Alex',
    text: 'First <reply>'
  }, {
    author: 'Sam',
    text: 'Second & reply'
  }];
  this.innerHTML = `<article>
        <button type="button" aria-expanded="${Comments.$$escapeHtml(`${open.get()}`, true)}">
          ${Comments.$$wccChild(label.get())}
        </button>
        <ul style="${Comments.$$escapeHtml(`${visibility.get()}`, true)}">
          ${replies.map(({author, text}, index) => `<li data-index="${Comments.$$escapeHtml(`${index}`, true)}">
              <h3>${Comments.$$wccChild(author)}</h3>
              <p>${Comments.$$wccChild(text)}</p>
            </li>`).join('')}
        </ul>
      </article>`;
  {
    const $evRoot0 = this.querySelector('article:nth-of-type(1)');
    const $evEl0 = $evRoot0?.querySelector(':scope > button:nth-of-type(1)');
    $evEl0?.addEventListener('click', event => this.toggle(event));
  }
}
