class First extends HTMLElement {
  static $$wccChild(value) {
    if (value == null || typeof value === 'boolean') return '';
    if (Array.isArray(value)) return value.map(First.$$wccChild).join('');
    return `${value}`.replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;');
  }
  connectedCallback() {
    this.render();
  }
  render() {
    this.innerHTML = `<ul>
        ${['first'].map(text => `<li>${First.$$wccChild(text)}</li>`).join('')}
      </ul>`;
  }
}
